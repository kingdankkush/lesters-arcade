import fs from 'node:fs';

const sources={},edges=[];
const sha=body=>createHash('sha256').update(body).digest('hex');
function visit(file){
  if(sources[file])return;const bytes=fs.readFileSync(path.join(repo,file));sources[file]={bytes:bytes.length,sha256:sha(bytes)};
  const text=bytes.toString('utf8');
  if(/\bimport\s*\(\s*(?!['"\s])/.test(text))throw new Error(`undeclared computed import:${file}`);
  for(const match of text.matchAll(/(?:\bfrom\s*|\bimport\s*\()\s*['"]([^'"]+)['"]/g)){
    const specifier=match[1];if(specifier.startsWith('node:'))continue;if(!specifier.startsWith('.'))throw new Error(`undeclared package:${specifier}`);
    const resolved=path.relative(repo,path.resolve(repo,path.dirname(file),specifier)).replaceAll('\\','/');
    if(resolved.startsWith('../'))throw new Error('source closure escaped own checkout');edges.push({file,specifier,resolved});visit(resolved);
  }
}
roots.forEach(visit);
