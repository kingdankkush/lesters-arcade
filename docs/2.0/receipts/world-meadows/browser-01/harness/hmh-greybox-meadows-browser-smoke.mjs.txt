async function walkTo(page,target,record){
  const before=await snapshot(page),begin=Date.now(),deadline=begin+Math.hypot(target.x-before.playtest.view.x,target.y-before.playtest.view.y)/140*1000+4000,held=new Set();
  try{
    while(true){
      const current=await snapshot(page),v=current.playtest.view,dx=target.x-v.x,dy=target.y-v.y;
      assert.equal(v.inspectionJumps,0,'actual continuous native path, no inspection teleport');
      assert.equal(current.playtest.planView,false,'actor-scale Walk view');
      assert.equal(v.contacts,0,'declared path walking must not scrape a physical blocker');
      if(Math.hypot(dx,dy)<=20)break;
      assert.ok(Date.now()<deadline,`native route stalled before${target.name}`);
      const next=new Set();
      if(Math.abs(dx)>12)next.add(dx>0?'ArrowRight':'ArrowLeft');
      if(Math.abs(dy)>12)next.add(dy>0?'ArrowDown':'ArrowUp');
      for(const key of held)if(!next.has(key)){await page.keyboard.up(key);held.delete(key);}
      for(const key of next)if(!held.has(key)){await page.keyboard.down(key);held.add(key);}
      // Real browser/60Hz clock advances; this is control sampling, not a replaced clock.
      await page.waitForTimeout(75);
    }
  }finally{for(const key of held)await page.keyboard.up(key);}
  await page.waitForFunction(()=>{const p=window.__HMH_GREYBOX__.snapshot().playtest;return p.heldKeys===0&&p.view.vx===0&&p.view.vy===0;},null,{timeout:2500});
  const after=await snapshot(page);assert.ok(Math.hypot(target.x-after.playtest.view.x,target.y-after.playtest.view.y)<=32,`native arrival:${target.name}`);assert.equal(after.playtest.view.inspectionJumps,0);assert.ok(after.playtest.view.tick>before.playtest.view.tick);assert.ok(after.playtest.nativeEvents>before.playtest.nativeEvents);record.journeys.push({target,before,after,wholeNativeControlWallMs:Date.now()-begin,basis:'Read-only position feedback samples actual browser keyboard controls; no clock/state writes. No contact at sampled feedback points; this is not cumulative contact telemetry, exact centre-polyline following, combat, pacing or performance certification.'});
}
async function touchTo(page,session,direction,target,record){
  const before=await snapshot(page),button=await page.getByRole('button',{name:`Move${' '+direction}`}).boundingBox();assert.ok(button);
  const x=button.x+button.width/2,y=button.y+button.height/2;
  try{
    await session.send('Input.dispatchTouchEvent',{type:'touchStart',touchPoints:[{x,y,id:1}]});
    await page.waitForFunction(({axis,value,direction})=>{const v=window.__HMH_GREYBOX__.snapshot().playtest.view;return(direction==='left'||direction==='up')?v[axis]<=value:v[axis]>=value;},{...target,direction},{timeout:8000});
  }finally{await session.send('Input.dispatchTouchEvent',{type:'touchCancel',touchPoints:[]});}
  await page.waitForFunction(()=>{const p=window.__HMH_GREYBOX__.snapshot().playtest;return p.heldPointers===0&&p.view.vx===0&&p.view.vy===0;});
  const after=await snapshot(page);assert.equal(after.playtest.view.inspectionJumps,0);assert.equal(after.playtest.view.contacts,0);assert.ok(after.playtest.view.tick>before.playtest.view.tick+30);assert.ok(after.playtest.nativeEvents>before.playtest.nativeEvents);record.journeys.push({direction,target,before,after,basis:'Actual Chrome touch dispatch on the visible direction button, with touchCancel and no inspection jump. Windows phone framing proxy, not physical phone.'});
}
