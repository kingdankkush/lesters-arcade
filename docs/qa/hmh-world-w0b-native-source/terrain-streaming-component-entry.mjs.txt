const profile=RUNTIME_PERFORMANCE_PROFILES[new URLSearchParams(location.search).get('profile')==='mobile'?'mobile':'desktop'];
const app=new Application();await app.init({width:innerWidth,height:innerHeight,resolution:profile.id==='mobile'?1:devicePixelRatio,antialias:profile.antialias,background:0x101822,preference:'webgl'});
document.body.appendChild(app.canvas);app.ticker.stop();
const production=createWorldProductionLayers({ContainerClass:Container,GraphicsClass:Graphics,TilingSpriteClass:TilingSprite});app.stage.addChild(production.root);
const registry=createTerrainTileRegistry({TilingSpriteClass:TilingSprite});

const beforeWorld=JSON.stringify(LEVEL_ONE_WORLD);
const spriteCount=source=>{let count=0;const visit=root=>{for(const child of root.children??[]){if(child.texture?.source===source)count++;visit(child);}};visit(production.root);return count;};
const wrapped={cache:Assets.cache,async load(url){const texture=await Assets.load(url);loads.push({url,uid:texture.source.uid,pixelWidth:texture.source.pixelWidth,pixelHeight:texture.source.pixelHeight});return texture;},async unload(url){
 const texture=Assets.cache.get(url),source=texture?.source;
 const record={url,sourceUid:source?.uid,spriteReferencesBeforeUnload:source?spriteCount(source):null};
 if(!texture||record.spriteReferencesBeforeUnload!==0)throw new Error('Actual consumer remained before source unload');
 await Assets.unload(url);Object.assign(record,{cachePresentAfterUnload:Assets.cache.has(url),textureDestroyed:texture.destroyed,sourceDestroyed:source.destroyed});unloads.push(record);
}};
const streaming=await createTerrainAreaStreaming({world:LEVEL_ONE_WORLD,registry,worldProduction:production,bake,Assets:wrapped,profile,onError:error=>errors.push(String(error.message))});

const view=()=>({width:innerWidth,height:innerHeight});
const render=()=>{bake.render({worldProduction:production,world:LEVEL_ONE_WORLD,camera:current,view:view(),worldToScreen,queryGround:createLevelOneGroundQuery(),tick:0,performanceProfile:profile,terrainTiles:registry});app.render();};

window.terrainProof=Object.freeze({
 begin(camera){current=createCameraState(camera);streaming.update({camera:{...current},view:view()});return snapshot();},
 async settle(){await streaming.idle();render();return snapshot();},
 snapshot,
 async dispose(){await streaming.dispose();const final=snapshot();app.destroy(true,{children:true});return final;},
});
